/* ---------- A2 ↔ SM-2 köprüsü ---------- */
function a2ResolveCard(c){
  if(c && c.src==='a2'){
    const w=(typeof A2_WOERTER!=='undefined'&&A2_WOERTER[c.ai])||{};
    const K=(typeof A2_META!=='undefined'&&A2_META.kategorien[w.k])?A2_META.kategorien[w.k].name:'';
    let forms='';
    if(w.k==='n')      forms=w.pl?('Plural: '+w.pl):'';
    else if(w.k==='v') forms=w.vf||'';
    else               forms=K+(w.g?(' · '+w.g):'');
    return { art:(w.k==='n'?(w.ar||''):K), de:w.w||'', tr:w.tr||'', ex:w.s||'', extr:w.st||'', forms:forms };
  }
  return CURRICULUM[c.d].vocab[c.i];
}
function a2CardLabel(c){ return (c&&c.src==='a2') ? 'A2 Wortliste' : ('Gün '+c.d+' kelimesi'); }
